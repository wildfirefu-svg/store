```python
# migrations/001_unify_data_sources.py
import json
import sqlite3
from pathlib import Path

class DataMigration:
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path)
        self.conn.row_factory = sqlite3.Row

    def migrate_json_to_sqlite(self):
        """迁移所有 JSON 数据到 SQLite"""
        # 迁移歌诀数据
        self.migrate_gejue()
        # 迁移测试用例
        self.migrate_test_cases()
        # 迁移配置数据
        self.migrate_configs()

    def migrate_gejue(self):
        """迁移歌诀数据"""
        with open('knowledge-base/gejue_core.json', 'r', encoding='utf-8') as f:
            gejue_data = json.load(f)

        # 创建歌诀表（如果不存在）
        self.conn.execute('''
            CREATE TABLE IF NOT EXISTS gejue_v2 (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                content TEXT NOT NULL,
                category TEXT,
                source TEXT,
                version TEXT DEFAULT 'v2',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                UNIQUE(name, category)
            )
        ''')

        # 批量插入
        cursor = self.conn.cursor()
        for gejue in gejue_data['gejue']:
            try:
                cursor.execute('''
                    INSERT OR REPLACE INTO gejue_v2
                    (name, content, category, source, version)
                    VALUES (?, ?, ?, ?, ?)
                ''', (gejue['name'], gejue['content'],
                      gejue.get('category', ''), gejue.get('source', ''),
                      gejue.get('version', 'v2')))
            except sqlite3.IntegrityError:
                continue

        self.conn.commit()

    def migrate_configs(self):
        """迁移配置数据"""
        with open('config.json', 'r', encoding='utf-8') as f:
            config_data = json.load(f)

        self.conn.execute('''
            CREATE TABLE IF NOT EXISTS configs (
                key TEXT PRIMARY KEY,
                value TEXT NOT NULL,
                type TEXT DEFAULT 'string',
                description TEXT,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        cursor = self.conn.cursor()
        for key, value in config_data.items():
            value_type = type(value).__name__
            cursor.execute('''
                INSERT OR REPLACE INTO configs
                (key, value, type, description)
                VALUES (?, ?, ?, ?)
            ''', (key, str(value), value_type, f'Migrated from config.json'))

        self.conn.commit()
```

```python
# migrations/migration_manager.py
import sqlite3
from pathlib import Path
from typing import List, Optional

class MigrationManager:
    def __init__(self, db_path: str, migrations_dir: str = "migrations"):
        self.db_path = db_path
        self.migrations_dir = Path(migrations_dir)
        self.conn = sqlite3.connect(db_path)

    def init_schema(self):
        """初始化迁移系统表"""
        self.conn.execute('''
            CREATE TABLE IF NOT EXISTS schema_migrations (
                version INTEGER PRIMARY KEY,
                name TEXT NOT NULL,
                applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                rollback_sql TEXT
            )
        ''')
        self.conn.commit()

    def get_current_version(self) -> int:
        """获取当前数据库版本"""
        result = self.conn.execute(
            'SELECT MAX(version) FROM schema_migrations'
        ).fetchone()
        return result[0] if result[0] else 0

    def get_pending_migrations(self) -> List[Path]:
        """获取待执行的迁移文件"""
        current_version = self.get_current_version()
        all_migrations = sorted(self.migrations_dir.glob("*.py"))

        # 过滤出需要执行的迁移
        pending = []
        for migration_file in all_migrations:
            # 从文件名提取版本号: 001_add_users_table.py
            try:
                version = int(migration_file.stem.split('_')[0])
                if version > current_version:
                    pending.append(migration_file)
            except (ValueError, IndexError):
                continue

        return pending

    def apply_migration(self, migration_file: Path) -> bool:
        """执行单个迁移文件"""
        try:
            # 动态导入迁移模块
            spec = importlib.util.spec_from_file_location(
                migration_file.stem, migration_file
            )
            module = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(module)

            # 执行迁移
            migration = module.Migration(self.conn)
            migration.up()

            # 记录迁移
            version = int(migration_file.stem.split('_')[0])
            self.conn.execute('''
                INSERT INTO schema_migrations (version, name, rollback_sql)
                VALUES (?, ?, ?)
            ''', (version, migration_file.name, getattr(migration, 'down_sql', '')))

            self.conn.commit()
            return True
        except Exception as e:
            print(f"迁移失败: {migration_file.name}, 错误: {e}")
            self.conn.rollback()
            return False

    def migrate(self):
        """执行所有待执行的迁移"""
        self.init_schema()
        pending = self.get_pending_migrations()

        if not pending:
            print("数据库已是最新版本")
            return

        print(f"发现 {len(pending)} 个待执行的迁移")
        for migration_file in pending:
            print(f"执行迁移: {migration_file.name}")
            if self.apply_migration(migration_file):
                print(f"✓ {migration_file.name} 执行成功")
            else:
                print(f"✗ {migration_file.name} 执行失败")
                break

# 使用示例
if __name__ == "__main__":
    manager = MigrationManager('data/bazi.db')
    manager.migrate()
```

**3. 数据备份与恢复**

```python
# utils/backup_manager.py
import sqlite3
import shutil
from datetime import datetime
from pathlib import Path
import gzip
import os

class BackupManager:
    def __init__(self, db_path: str, backup_dir: str = "backups"):
        self.db_path = Path(db_path)
        self.backup_dir = Path(backup_dir)
        self.backup_dir.mkdir(exist_ok=True)

    def create_backup(self, compression: bool = True) -> Path:
        """创建数据库备份"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_name = f"bazi_backup_{timestamp}.db"

        if compression:
            backup_path = self.backup_dir / f"{backup_name}.gz"
            self._create_compressed_backup(backup_path)
        else:
            backup_path = self.backup_dir / backup_name
            self._create_standard_backup(backup_path)

        print(f"备份已创建: {backup_path}")
        return backup_path

    def _create_standard_backup(self, backup_path: Path):
        """创建标准备份"""
        # 使用 SQLite 的在线备份 API
        source = sqlite3.connect(str(self.db_path))
        backup = sqlite3.connect(str(backup_path))

        source.backup(backup)
        backup.close()
        source.close()

    def _create_compressed_backup(self, backup_path: Path):
        """创建压缩备份"""
        temp_backup = self.backup_dir / "temp_backup.db"
        self._create_standard_backup(temp_backup)

        # 压缩备份文件
        with open(temp_backup, 'rb') as f_in:
            with gzip.open(backup_path, 'wb') as f_out:
                shutil.copyfileobj(f_in, f_out)

        # 删除临时文件
        temp_backup.unlink()

    def restore_backup(self, backup_path: Path, validate: bool = True) -> bool:
        """恢复数据库备份"""
        try:
            # 验证备份文件
            if validate and not self._validate_backup(backup_path):
                print("备份文件验证失败")
                return False

            # 创建当前数据库的备份
            if self.db_path.exists():
                self.create_backup(compression=False)

            # 解压并恢复
            if backup_path.suffix == '.gz':
                temp_backup = self.backup_dir / "temp_restore.db"
                with gzip.open(backup_path, 'rb') as f_in:
                    with open(temp_backup, 'wb') as f_out:
                        shutil.copyfileobj(f_in, f_out)
                backup_path = temp_backup

            # 恢复数据
            backup = sqlite3.connect(str(backup_path))
            current = sqlite3.connect(str(self.db_path))
            backup.backup(current)

            current.close()
            backup.close()

            # 清理临时文件
            if temp_backup.exists():
                temp_backup.unlink()

            print(f"数据库已从 {backup_path} 恢复")
            return True

        except Exception as e:
            print(f"恢复失败: {e}")
            return False

    def _validate_backup(self, backup_path: Path) -> bool:
        """验证备份文件完整性"""
        try:
            if backup_path.suffix == '.gz':
                # 验证 gzip 文件
                with gzip.open(backup_path, 'rb') as f:
                    header = f.read(16)  # SQLite 文件头
                    return header.startswith(b'SQLite format 3')
            else:
                # 验证普通 SQLite 文件
                conn = sqlite3.connect(str(backup_path))
                conn.execute("SELECT name FROM sqlite_master WHERE type='table'")
                conn.close()
                return True
        except Exception:
            return False

    def list_backups(self) -> List[Path]:
        """列出所有备份文件"""
        backups = list(self.backup_dir.glob("bazi_backup_*"))
        return sorted(backups, key=lambda x: x.stat().st_mtime, reverse=True)

    def cleanup_old_backups(self, keep_count: int = 5):
        """清理旧备份，保留最近的几个"""
        backups = self.list_backups()
        if len(backups) > keep_count:
            for old_backup in backups[keep_count:]:
                old_backup.unlink()
                print(f"已删除旧备份: {old_backup}")

# 定时备份任务
def schedule_daily_backup(db_path: str):
    """配置每日自动备份"""
    backup_manager = BackupManager(db_path)

    # 每天凌晨 2 点执行备份
    schedule.every().day.at("02:00").do(
        backup_manager.create_backup,
        compression=True
    )

    # 每周清理旧备份，保留最近 7 天的
    schedule.every().week.do(
        backup_manager.cleanup_old_backups,
        keep_count=7
    )
```

```python
# utils/database_optimizer.py
import sqlite3
from typing import List, Dict

class DatabaseOptimizer:
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path)

    def analyze_indexes(self) -> List[Dict]:
        """分析当前索引使用情况"""
        # 获取所有索引
        indexes = self.conn.execute('''
            SELECT name, tbl_name, sql
            FROM sqlite_master
            WHERE type='index' AND name NOT LIKE 'sqlite_%'
        ''').fetchall()

        analysis = []
        for index in indexes:
            # 获取索引统计信息
            stats = self._get_index_stats(index[0])
            analysis.append({
                'name': index[0],
                'table': index[1],
                'sql': index[2],
                'stats': stats
            })

        return analysis

    def _get_index_stats(self, index_name: str) -> Dict:
        """获取索引统计信息"""
        try:
            # 获取索引大小
            size = self.conn.execute(f'''
                PRAGMA index_info({index_name})
            ''').fetchall()

            # 获取索引使用情况
            usage = self.conn.execute(f'''
                ANALYZE {index_name}
            ''').fetchall()

            return {'size': len(size), 'usage': len(usage)}
        except Exception:
            return {'size': 0, 'usage': 0}

    def create_recommended_indexes(self):
        """创建推荐的索引"""
        recommended_indexes = [
            # 歌诀全文搜索索引
            ('idx_gejue_fts', 'gejue', '''
                CREATE VIRTUAL TABLE IF NOT EXISTS gejue_fts USING fts5(
                    name, content, category,
                    content='gejue',
                    content_rowid='id'
                )
            '''),

            # 案例检索索引
            ('idx_case_vector', 'cases', '''
                CREATE INDEX IF NOT EXISTS idx_case_vector
                ON cases(embedding_vector)
            '''),

            # 时间范围查询索引
            ('idx_charts_created', 'charts', '''
                CREATE INDEX IF NOT EXISTS idx_charts_created
                ON charts(created_at DESC)
            '''),

            # 用户查询索引
            ('idx_users_email', 'users', '''
                CREATE INDEX IF NOT EXISTS idx_users_email
                ON users(email)
            '''),
        ]

        for index_name, table, sql in recommended_indexes:
            try:
                self.conn.execute(sql)
                print(f"✓ 创建索引: {index_name}")
            except Exception as e:
                print(f"✗ 创建索引失败: {index_name}, {e}")

        self.conn.commit()

    def optimize_database(self):
        """优化数据库"""
        print("开始数据库优化...")

        # 分析表统计信息
        self.conn.execute("ANALYZE")
        print("✓ 分析完成")

        # 重建数据库（整理碎片）
        self.conn.execute("VACUUM")
        print("✓ 碎片整理完成")

        # 优化索引
        self.create_recommended_indexes()
        print("✓ 索引优化完成")

        # 设置性能参数
        self.conn.execute("PRAGMA journal_mode=WAL")
        self.conn.execute("PRAGMA synchronous=NORMAL")
        self.conn.execute("PRAGMA cache_size=-64000")  # 64MB 缓存
        self.conn.execute("PRAGMA temp_store=MEMORY")
        self.conn.commit()

        print("数据库优化完成")

    def get_slow_queries(self, threshold_ms: int = 100) -> List[Dict]:
        """获取慢查询日志"""
        # SQLite 本身不记录慢查询，需要应用层实现
        # 这里提供一个实现思路
        self.conn.execute('''
            CREATE TABLE IF NOT EXISTS slow_queries (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                query TEXT NOT NULL,
                execution_time REAL NOT NULL,
                timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')

        slow_queries = self.conn.execute('''
            SELECT query, execution_time, timestamp
            FROM slow_queries
            WHERE execution_time > ?
            ORDER BY execution_time DESC
            LIMIT 20
        ''', (threshold_ms,)).fetchall()

        return [{
            'query': q[0],
            'time': q[1],
            'timestamp': q[2]
        } for q in slow_queries]

# 使用示例
if __name__ == "__main__":
    optimizer = DatabaseOptimizer('data/bazi.db')
    optimizer.optimize_database()
```

### 1.2 缓存策略优化

#### 当前问题分析

```python
# cache/redis_cache.py
import redis
import json
import pickle
from typing import Any, Optional, List
from datetime import timedelta
import hashlib

class RedisCacheManager:
    def __init__(
        self,
        host: str = 'localhost',
        port: int = 6379,
        db: int = 0,
        password: Optional[str] = None,
        default_ttl: int = 3600
    ):
        self.redis = redis.Redis(
            host=host,
            port=port,
            db=db,
            password=password,
            decode_responses=False  # 支持二进制数据
        )
        self.default_ttl = default_ttl

    def get(self, key: str) -> Optional[Any]:
        """获取缓存数据"""
        try:
            cached = self.redis.get(key)
            if cached:
                return pickle.loads(cached)
            return None
        except Exception as e:
            print(f"缓存读取失败: {e}")
            return None

    def set(
        self,
        key: str,
        value: Any,
        ttl: Optional[int] = None
    ) -> bool:
        """设置缓存数据"""
        try:
            ttl = ttl or self.default_ttl
            serialized = pickle.dumps(value)
            return self.redis.setex(key, ttl, serialized)
        except Exception as e:
            print(f"缓存写入失败: {e}")
            return False

    def delete(self, key: str) -> bool:
        """删除缓存"""
        try:
            return bool(self.redis.delete(key))
        except Exception as e:
            print(f"缓存删除失败: {e}")
            return False

    def delete_pattern(self, pattern: str) -> int:
        """批量删除缓存"""
        try:
            keys = self.redis.keys(pattern)
            if keys:
                return self.redis.delete(*keys)
            return 0
        except Exception as e:
            print(f"批量删除失败: {e}")
            return 0

    def get_multi(self, keys: List[str]) -> dict:
        """批量获取缓存"""
        try:
            values = self.redis.mget(keys)
            result = {}
            for key, value in zip(keys, values):
                if value:
                    result[key] = pickle.loads(value)
            return result
        except Exception as e:
            print(f"批量获取失败: {e}")
            return {}

    def set_multi(self, mapping: dict, ttl: Optional[int] = None) -> bool:
        """批量设置缓存"""
        try:
            pipe = self.redis.pipeline()
            ttl = ttl or self.default_ttl

            for key, value in mapping.items():
                serialized = pickle.dumps(value)
                pipe.setex(key, ttl, serialized)

            pipe.execute()
            return True
        except Exception as e:
            print(f"批量设置失败: {e}")
            return False

    def exists(self, key: str) -> bool:
        """检查缓存是否存在"""
        try:
            return bool(self.redis.exists(key))
        except Exception:
            return False

    def increment(self, key: str, amount: int = 1) -> int:
        """递增计数器"""
        try:
            return self.redis.incr(key, amount)
        except Exception:
            return 0

    def get_stats(self) -> dict:
        """获取缓存统计信息"""
        try:
            info = self.redis.info()
            return {
                'used_memory': info['used_memory_human'],
                'used_memory_peak': info['used_memory_peak_human'],
                'total_connections_received': info['total_connections_received'],
                'total_commands_processed': info['total_commands_processed'],
                'keyspace_hits': info['keyspace_hits'],
                'keyspace_misses': info['keyspace_misses'],
                'hit_rate': self._calculate_hit_rate(info)
            }
        except Exception as e:
            print(f"获取统计信息失败: {e}")
            return {}

    def _calculate_hit_rate(self, info: dict) -> float:
        """计算缓存命中率"""
        hits = info.get('keyspace_hits', 0)
        misses = info.get('keyspace_misses', 0)
        total = hits + misses
        if total == 0:
            return 0.0
        return (hits / total) * 100

    def flush_db(self) -> bool:
        """清空当前数据库"""
        try:
            self.redis.flushdb()
            return True
        except Exception as e:
            print(f"清空数据库失败: {e}")
            return False
```

**2. 多级缓存管理器**

```python
# cache/multi_level_cache.py
from functools import wraps
from typing import Any, Optional, Callable
import hashlib
import json

class MultiLevelCache:
    def __init__(
        self,
        redis_cache: Optional[RedisCacheManager] = None,
        l1_size: int = 1000,
        l2_ttl: int = 3600,
        l3_ttl: int = 86400
    ):
        self.redis = redis_cache
        self.l1_cache = {}  # 内存缓存 L1
        self.l1_size = l1_size
        self.l2_ttl = l2_ttl  # Redis 缓存时间
        self.l3_ttl = l3_ttl  # 持久化缓存时间

    def _generate_key(self, func_name: str, args: tuple, kwargs: dict) -> str:
        """生成缓存键"""
        # 将参数转换为字符串
        key_data = {
            'func': func_name,
            'args': str(args),
            'kwargs': str(sorted(kwargs.items()))
        }
        key_string = json.dumps(key_data, sort_keys=True)
        return hashlib.md5(key_string.encode()).hexdigest()

    def get(self, key: str) -> Optional[Any]:
        """多级缓存获取"""
        # L1 缓存（内存）
        if key in self.l1_cache:
            return self.l1_cache[key]

        # L2 缓存（Redis）
        if self.redis:
            value = self.redis.get(key)
            if value:
                # 回填 L1 缓存
                self.l1_cache[key] = value
                return value

        return None

    def set(self, key: str, value: Any, level: str = 'l2') -> bool:
        """多级缓存设置"""
        try:
            # L1 缓存
            if len(self.l1_cache) >= self.l1_size:
                # LRU 淘汰
                oldest_key = next(iter(self.l1_cache))
                del self.l1_cache[oldest_key]
            self.l1_cache[key] = value

            # L2 缓存（Redis）
            if self.redis and level in ['l2', 'l3']:
                ttl = self.l2_ttl if level == 'l2' else self.l3_ttl
                return self.redis.set(key, value, ttl)

            return True
        except Exception as e:
            print(f"缓存设置失败: {e}")
            return False

    def invalidate(self, key: str):
        """使缓存失效"""
        # 清除 L1 缓存
        if key in self.l1_cache:
            del self.l1_cache[key]

        # 清除 L2 缓存
        if self.redis:
            self.redis.delete(key)

    def invalidate_pattern(self, pattern: str):
        """批量使缓存失效"""
        # 清除 L1 缓存
        keys_to_delete = [
            key for key in self.l1_cache.keys()
            if pattern in key
        ]
        for key in keys_to_delete:
            del self.l1_cache[key]

        # 清除 L2 缓存
        if self.redis:
            self.redis.delete_pattern(pattern)

# 缓存装饰器
def cached(
    cache_manager: MultiLevelCache,
    ttl: int = 3600,
    key_prefix: str = ""
):
    """缓存装饰器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 生成缓存键
            cache_key = f"{key_prefix}_{cache_manager._generate_key(func.__name__, args, kwargs)}"

            # 尝试从缓存获取
            cached_value = cache_manager.get(cache_key)
            if cached_value is not None:
                return cached_value

            # 执行函数
            result = func(*args, **kwargs)

            # 存入缓存
            cache_manager.set(cache_key, result, level='l2')

            return result
        return wrapper
    return decorator
```

```python
# ai/prompt_versioning.py
import hashlib
from datetime import datetime

class PromptVersionManager:
    """Prompt 版本管理器"""
    
    def __init__(self, db_conn):
        self.db = db_conn
        
    def register_prompt(self, name: str, content: str, author: str) -> str:
        """注册新版本 Prompt"""
        version_id = hashlib.md5(f"{name}:{content}:{datetime.now()}".encode()).hexdigest()[:8]
        self.db.execute('''
            INSERT INTO prompt_versions (version_id, name, content, author, created_at)
            VALUES (?, ?, ?, ?, ?)
        ''', (version_id, name, content, author, datetime.now()))
        self.db.commit()
        return version_id
        
    def get_ab_test_prompt(self, test_name: str, user_id: str) -> str:
        """根据用户 ID 哈希分配 A/B 测试组"""
        # 简单的一致性哈希分组
        group = hash(user_id) % 2
        if group == 0:
            return self._get_prompt_version(test_name, 'version_A')
        else:
            return self._get_prompt_version(test_name, 'version_B')
```

```python
# ai/consensus_engine.py
import asyncio

class ConsensusEngine:
    """多模型共识引擎，用于关键结论的防幻觉校验"""
    
    async def verify_critical_claim(self, claim: str, chart_data: Dict) -> bool:
        """针对关键论断（如：寿命、重大灾祸），调用多个模型进行交叉验证"""
        models = ["claude-3-5-sonnet", "deepseek-r1", "qwen-max"]
        tasks = [self._ask_model(m, f"验证以下命理结论是否合理：{claim}，命盘：{chart_data}") for m in models]
        results = await asyncio.gather(*tasks)
        
        # 如果多数模型认为不合理，则触发警告或重写
        disagree_count = sum(1 for r in results if "不合理" in r or "错误" in r)
        return disagree_count < len(models) // 2
```
