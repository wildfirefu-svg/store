```javascript
// MingzhuManager: stores/loads chart data in sessionStorage
const MingzhuManager = {
    STORAGE_KEY: 'bazi_mingzhu_list',
    
    getAll() {
        try { return JSON.parse(sessionStorage.getItem(this.STORAGE_KEY)) || []; }
        catch { return []; }
    },
    
    save(mingzhu) {
        const list = this.getAll();
        // Deduplicate by chart_id
        const idx = list.findIndex(m => m.chart_id === mingzhu.chart_id);
        if (idx >= 0) list[idx] = mingzhu;
        else list.push(mingzhu);
        if (list.length > 6) list.shift(); // max 6
        sessionStorage.setItem(this.STORAGE_KEY, JSON.stringify(list));
        return list;
    },
    
    remove(chartId) {
        const list = this.getAll().filter(m => m.chart_id !== chartId);
        sessionStorage.setItem(this.STORAGE_KEY, JSON.stringify(list));
        return list;
    },
    
    getCurrent() {
        const id = sessionStorage.getItem('bazi_current_mingzhu');
        if (!id) return null;
        return this.getAll().find(m => m.chart_id === id) || null;
    },
    
    setCurrent(chartId) {
        sessionStorage.setItem('bazi_current_mingzhu', chartId);
    }
};
```

**Module B: ChatEngine** — handles messages + API calls

```javascript
const ChatEngine = {
    async sendMessage(text) {
        const current = MingzhuManager.getCurrent();
        const chartId = current ? current.chart_id : null;
        
        // Add user message to chat
        UI.addChatMessage('user', text);
        
        // If it looks like birth info and no chart yet, auto-create chart
        if (!chartId && this._looksLikeBirthInfo(text)) {
            const birth = this._parseBirthInfo(text);
            if (birth) {
                UI.addChatMessage('agent', '正在为您排盘...', '🔧');
                const chart = await this._createChart(birth);
                if (chart) {
                    const mz = {
                        chart_id: chart.chart_id,
                        name: birth.name || '命主',
                        birth: `${birth.year}-${String(birth.month).padStart(2,'0')}-${String(birth.day).padStart(2,'0')}`,
                        gender: birth.gender,
                        day_master: chart.day_master.gan + chart.day_master.wuxing,
                        summary: this._summarizeChart(chart)
                    };
                    MingzhuManager.save(mz);
                    MingzhuManager.setCurrent(chart.chart_id);
                    UI.refreshMingzhuPanel();
                    UI.renderChartColumns(chart);
                    
                    // Generate initial novice report
                    const report = await this._generateReport(chart.chart_id, 5);
                    UI.showReport('overview', report);
                    
                    UI.addChatMessage('agent', 
                        `排盘完毕。日主<b>${chart.day_master.gan}${chart.day_master.wuxing}</b>，格局<b>食神格</b>。整体来看，您聪明敏感，适合技术或文化行业。您可以继续问我任何问题。`);
                    return;
                }
            }
        }
        
        if (!chartId) {
            UI.addChatMessage('agent', '请先输入出生信息，例如：<b>公历 1993年7月15日 14:30 男 北京</b>');
            return;
        }
        
        // Route to /api/chat/stream for SSE streaming
        UI.setReportStatus('⏳ 正在分析...');
        this._streamChatAPI(chartId, text);
    },
    
    _looksLikeBirthInfo(text) {
        return /\d{4}.*\d{1,2}.*\d{1,2}/.test(text) || /农历|公历/.test(text);
    },
    
    _parseBirthInfo(text) {
        // Extract year/month/day/hour/gender from text
        const m = text.match(/(\d{4})[年\-\/\s]*(\d{1,2})[月\-\/\s]*(\d{1,2})/);
        if (!m) return null;
        const hourM = text.match(/(\d{1,2})[：:]\s*(\d{2})/);
        const genderM = /[男male]/.test(text) ? 'male' : /[女female]/.test(text) ? 'female' : null;
        return {
            year: parseInt(m[1]), month: parseInt(m[2]), day: parseInt(m[3]),
            hour: hourM ? parseInt(hourM[1]) : 12,
            minute: hourM ? parseInt(hourM[2]) : 0,
            gender: genderM || 'male',
            location: '北京',
            name: text.split(/[，,\s]/)[0] || ''
        };
    },
    
    async _createChart(birth) {
        const r = await fetch('/api/chart', {
            method: 'POST', headers: {'Content-Type':'application/json'},
            body: JSON.stringify(birth)
        });
        return r.ok ? r.json() : null;
    },
    
    async _generateReport(chartId, mode) {
        const r = await fetch('/api/analyze', {
            method: 'POST', headers: {'Content-Type':'application/json'},
            body: JSON.stringify({chart_id: chartId, mode: mode})
        });
        return r.ok ? (await r.json()).report : null;
    },
    
    _streamChatAPI(chartId, message) {
        // SSE streaming: progressively update chat reply + report
        const params = new URLSearchParams({chart_id: chartId, message: message});
        const evtSource = new EventSource(`/api/chat/stream?${params}`);
        
        let replyText = '';
        let reportText = '';
        let reportTab = 'overview';
        let toolUsed = null;
        const replyMsgEl = UI.addChatMessage('agent', '', null);
        const replyBubble = replyMsgEl.querySelector('.bubble');
        
        evtSource.addEventListener('token', (e) => {
            const data = JSON.parse(e.data);
            if (data.type === 'reply') {
                replyText += data.text;
                replyBubble.innerHTML = replyText;
                UI.scrollChatBottom();
            } else if (data.type === 'report') {
                reportText += data.text;
                reportTab = data.tab || reportTab;
                UI.showReportStreaming(reportTab, reportText);
            } else if (data.type === 'tool') {
                toolUsed = data.name;
            }
        });
        
        evtSource.addEventListener('done', (e) => {
            const data = JSON.parse(e.data);
            if (toolUsed) {
                const sender = replyMsgEl.querySelector('.sender');
                sender.innerHTML = '玄机子 <span class="tool-tag">🔧 ' + toolUsed + '</span>';
            }
            if (reportText) {
                UI.showReport(reportTab, reportText);
                UI.setReportStatus('✅ 已修正' + (data.corrections || 0) + '处');
            }
            UI.setReportStatus('');
            evtSource.close();
        });
        
        evtSource.addEventListener('error', () => {
            evtSource.close();
            if (!replyText) replyBubble.innerHTML = '分析服务暂时不可用，请稍后重试。';
        });
    },
    
    _summarizeChart(chart) {
        const dm = chart.day_master;
        const dy = (chart.da_yun||[]).find(d => d.is_current);
        return {
            day_master: dm.gan + dm.wuxing,
            current_dayun: dy ? dy.gan + dy.zhi : '',
            wuxing: chart.wuxing_stats
        };
    }
};
```

```javascript
const UI = {
```

```python
@app.get("/api/chat/stream")
async def chat_stream(chart_id: str, message: str):
    """SSE streaming chat — progressively sends reply tokens + report content."""
    chart = chart_cache._cache.get(chart_id)
    if not chart:
        async def error_stream():
            yield f"data: {json.dumps({'type':'reply','text':'请先提供出生信息进行排盘。'})}\n\n"
            yield f"data: {json.dumps({'type':'done'})}\n\n"
        return StreamingResponse(error_stream(), media_type="text/event-stream")

    msg = message
    tool_used = None
    report_content = ""
    report_tab = "overview"

    # ---- Build the response generator ----
    async def event_stream():
        nonlocal tool_used, report_content, report_tab

        # Step 1: Send tool detection
        if any(kw in msg for kw in ['财运','发财','投资','赚钱','破财']):
            tool_used = "流年分析"
            yield _sse_event('tool', {'name': tool_used})
            yield _sse_event('reply', {'text': '正在分析您的财运...\n\n'})
            await asyncio.sleep(0.1)

            # Generate liunian data
            birth = chart.get('birth_info', {})
            today = date.today()
            lc = _import_tool('liunian_calendar', 'knowledge-base/liunian_calendar.py')
            cal = lc.generate_year_calendar(
                birth.get('year', today.year-35), birth.get('month',1), birth.get('day',1),
                birth.get('hour',8), birth.get('gender','male'), today.year, chart
            )
            best = cal['overview']['best_month']
            worst = cal['overview']['worst_month']

            # Stream report token by token
            report_content = f"# 财运专题分析\n\n年均财运评分: {cal['overview']['avg_scores']['wealth']}/5\n\n"
            report_content += f"**最佳月份**: {best['month']}月 ({best['rating']})\n\n"
            report_content += f"**需注意**: {worst['month']}月 ({worst['rating']})\n\n"
            for m in cal['months']:
                report_content += f"- {m['month']}月 {m['ganzhi']} | 财运{m['wealth']['score']}⭐ | {m['wealth']['notes']}\n"

            report_tab = "财运专题"
            # Stream reply tokens with typewriter effect
            reply = f"已为您生成财运专题报告（见右侧）。今年最佳财运月在{best['month']}月，需注意{worst['month']}月。"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)
            # Stream report
            for i in range(0, len(report_content), 3):
                yield _sse_event('report', {'text': report_content[i:i+3], 'tab': report_tab})
                await asyncio.sleep(0.005)

        elif any(kw in msg for kw in ['感情','婚姻','结婚','配偶','恋爱','桃花']):
            tool_used = "命盘分析"
            yield _sse_event('tool', {'name': tool_used})
            dm = chart['day_master']
            fp = chart['four_pillars']
            day_zhi = fp['day']['zhi']
            reply = f"您的日支为{day_zhi}，是配偶宫。日主{dm.get('gan','')}{dm.get('wuxing','')}，感情方面建议稳定发展。"
            report_content = f"# 感情专题分析\n\n日支(配偶宫): **{day_zhi}**\n\n日主: **{dm.get('gan','')}{dm.get('wuxing','')}**\n\n> 您可以追问更具体的问题。"
            report_tab = "感情专题"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)
            for i in range(0, len(report_content), 3):
                yield _sse_event('report', {'text': report_content[i:i+3], 'tab': report_tab})
                await asyncio.sleep(0.005)

        elif any(kw in msg for kw in ['事业','工作','跳槽','升职','创业']):
            tool_used = "流年分析"
            yield _sse_event('tool', {'name': tool_used})
            birth = chart.get('birth_info', {})
            today = date.today()
            lc = _import_tool('liunian_calendar', 'knowledge-base/liunian_calendar.py')
            cal = lc.generate_year_calendar(
                birth.get('year', today.year-35), birth.get('month',1), birth.get('day',1),
                birth.get('hour',8), birth.get('gender','male'), today.year, chart
            )
            reply = f"已生成事业专题。年均评分{cal['overview']['avg_scores']['career']}/5。"
            report_content = f"# 事业专题分析\n\n年均事业评分: {cal['overview']['avg_scores']['career']}/5\n\n"
            for m in cal['months']:
                report_content += f"- {m['month']}月 | 事业{m['career']['score']}⭐ | {m['career']['notes']}\n"
            report_tab = "事业专题"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)
            for i in range(0, len(report_content), 3):
                yield _sse_event('report', {'text': report_content[i:i+3], 'tab': report_tab})
                await asyncio.sleep(0.005)

        elif any(kw in msg for kw in ['深度','四合出','四派','完整报告']):
            tool_used = "四合出分析"
            yield _sse_event('tool', {'name': tool_used})
            conclusions = _auto_analyze(chart)
            rb = _import_tool('report_builder', 'report_builder.py')
            report_content = rb.build_mode5_report(chart, conclusions)
            report_tab = "四合出"
            reply = "已为您生成四合出深度报告（见右侧）。四体系交叉验证，结论更可靠。"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)
            for i in range(0, len(report_content), 5):
                yield _sse_event('report', {'text': report_content[i:i+5], 'tab': report_tab})
                await asyncio.sleep(0.003)

        elif any(kw in msg for kw in ['名字','取名','改名']):
            tool_used = "取名分析"
            yield _sse_event('tool', {'name': tool_used})
            na = _import_tool('name_analysis', 'knowledge-base/name_analysis.py')
            names = na.generate_names('张', chart, 'male', 5)
            reply = f"为您推荐了5个匹配八字的名字，最佳: {names[0]['name']}({names[0]['total_score']}分)"
            report_content = f"# 取名推荐\n\n"
            for i, n in enumerate(names):
                report_content += f"{i+1}. **{n['name']}** — {n['total_score']}分 {n['grade']}\n\n"
            report_tab = "取名推荐"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)
            for i in range(0, len(report_content), 3):
                yield _sse_event('report', {'text': report_content[i:i+3], 'tab': report_tab})

        else:
            reply = "您可以问我关于财运、事业、感情、健康等具体问题，或者说"深度报告"获取四合出完整分析。"
            for i in range(1, len(reply)+1, 2):
                yield _sse_event('reply', {'text': reply[i-1:i+1]})
                await asyncio.sleep(0.02)

        yield _sse_event('done', {'corrections': 0})

    return StreamingResponse(event_stream(), media_type="text/event-stream",
                             headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

def _sse_event(event_type, data):
    return f"event: {event_type}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"
```
